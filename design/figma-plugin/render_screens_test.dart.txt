/// The app's icon font, Material Symbols Rounded, from the
/// material_symbols_icons package in the pub cache (where
/// .dart_tool/package_config.json says it is).
String symbolsFont() {
  final config = File(
    '${Directory.current.path}/.dart_tool/package_config.json',
  );
  final packages =
      (jsonDecode(config.readAsStringSync()) as Map<String, dynamic>)['packages']
          as List;
  final package = packages.cast<Map<String, dynamic>>().firstWhere(
    (p) => p['name'] == 'material_symbols_icons',
  );
  // A folder: the trailing slash keeps its last segment when resolving.
  final root = Uri.file(config.path).resolve('${package['rootUri']}/');
  return root.resolve('lib/fonts/MaterialSymbolsRounded.ttf').toFilePath();
}

final written = <String>[];

/// Writes what is on screen to `<outDir>/<name>.png` at 2x.
Future<void> shot(WidgetTester tester, String name) async {
  await tester.pumpAndSettle();
  // Pictures decode outside the test's fake clock: give them real time.
  if (find.byType(Image).evaluate().isNotEmpty) {
    for (var i = 0; i < 6; i++) {
      await tester.runAsync(
        () => Future<void>.delayed(const Duration(milliseconds: 20)),
      );
      await tester.pump();
    }
    await tester.pumpAndSettle();
  }
  final view = tester.binding.renderViews.first;
  final layer = view.debugLayer! as OffsetLayer;
  final bounds = view.paintBounds;
  final bytes = await tester.runAsync(() async {
    final image = await layer.toImage(bounds);
    try {
      final data = await image.toByteData(format: ui.ImageByteFormat.png);
      return data!.buffer.asUint8List();
    } finally {
      image.dispose();
    }
  });
  File('$outDir/$name.png').writeAsBytesSync(bytes!);
  written.add(name);
}

/// How much taller the content of the vertical lists on screen is than the
/// lists: positive when one scrolls, negative when they all have room to
/// spare (then the one closest to scrolling counts).
double overflow(WidgetTester tester) {
  final size = tester.view.physicalSize / tester.view.devicePixelRatio;
  final visible = Offset.zero & size;
  double? most;
  for (final element in find.byType(Scrollable).evaluate()) {
    final state = (element as StatefulElement).state as ScrollableState;
    if (state.widget.axis != Axis.vertical) continue;
    final box = element.findRenderObject() as RenderBox?;
    if (box == null || !box.attached || !box.hasSize) continue;
    final rect = box.localToGlobal(Offset.zero) & box.size;
    if (!visible.overlaps(rect)) continue;
    final position = state.position;
    if (!position.hasContentDimensions) continue;
    final viewport = viewportOf(box, position);
    if (viewport is RenderShrinkWrappingViewport) continue;
    final double slack;
    if (viewport == null) {
      // A single child (SingleChildScrollView): only how far it scrolls.
      slack = position.maxScrollExtent - position.minScrollExtent;
    } else {
      var content = 0.0;
      for (final sliver in viewport.childrenInPaintOrder) {
        content += sliver.geometry?.scrollExtent ?? 0;
      }
      slack = content - position.viewportDimension;
    }
    most = most == null ? slack : math.max(most, slack);
  }
  return most ?? 0;
}

/// The viewport that [position] scrolls, under [root].
RenderViewportBase? viewportOf(RenderObject root, ScrollPosition position) {
  RenderViewportBase? found;
  void visit(RenderObject child) {
    if (found != null) return;
    if (child is RenderViewportBase && identical(child.offset, position)) {
      found = child;
      return;
    }
    child.visitChildren(visit);
  }

  visit(root);
  return found;
}

/// [shot], then, when the screen scrolls, `<name>_full`: the whole page on a
/// phone just tall enough for it.
Future<void> shotFull(WidgetTester tester, String name) async {
  await shot(tester, name);
  if (overflow(tester) < 1) return;
  var height = screen.height;
  for (var i = 0; i < 12; i++) {
    final slack = overflow(tester);
    if (slack.abs() < 1) break;
    final next = math.max(screen.height, (height + slack).ceilToDouble());
    if (next == height) break;
    height = next;
    tester.view.physicalSize = Size(screen.width, height) * pixelRatio;
    await tester.pumpAndSettle();
  }
  await shot(tester, '${name}_full');
  tester.view.physicalSize = screen * pixelRatio;
  await tester.pumpAndSettle();
}

/// Taps [finder], scrolling to it only when it is out of sight.
Future<void> tapAndSettle(WidgetTester tester, Finder finder) async {
  if (finder.hitTestable().evaluate().isEmpty) {
    await tester.ensureVisible(finder);
    await tester.pumpAndSettle();
  }
  await tester.tap(finder);
  await tester.pumpAndSettle();
}

/// A test with real shadows: flutter_test draws them as solid outlines, and
/// wants that back before the test ends.
void screenTest(
  String description,
  Future<void> Function(WidgetTester tester) body,
) {
  testWidgets(description, (tester) async {
    debugDisableShadows = false;
    try {
      await body(tester);
    } finally {
      debugDisableShadows = true;
    }
  });
}

Future<Uint8List> picture(
  int width,
  int height,
  void Function(Canvas canvas) paint,
) async {
  final recorder = ui.PictureRecorder();
  paint(Canvas(recorder));
  final recorded = recorder.endRecording();
  final image = await recorded.toImage(width, height);
  final data = await image.toByteData(format: ui.ImageByteFormat.png);
  image.dispose();
  recorded.dispose();
  return data!.buffer.asUint8List();
}

void write(
  Canvas canvas,
  String value,
  double x,
  double y, {
  double size = 15,
  FontWeight weight = FontWeight.w400,
  Color color = ink,
  double width = 520,
  TextAlign align = TextAlign.left,
}) {
  final builder =
      ui.ParagraphBuilder(
          ui.ParagraphStyle(
            textAlign: align,
            fontFamily: 'Arimo',
            fontSize: size,
            fontWeight: weight,
          ),
        )
        ..pushStyle(
          ui.TextStyle(
            color: color,
            fontFamily: 'Arimo',
            fontSize: size,
            fontWeight: weight,
          ),
        )
        ..addText(value);
  final paragraph = builder.build()
    ..layout(ui.ParagraphConstraints(width: width));
  canvas.drawParagraph(paragraph, Offset(x, y));
}

void line(
  Canvas canvas,
  Offset from,
  Offset to, {
  Color color = const Color(0x40000000),
  double width = 1.5,
}) => canvas.drawLine(
  from,
  to,
  Paint()
    ..color = color
    ..strokeWidth = width,
);

void fill(Canvas canvas, Rect rect, Color color) =>
    canvas.drawRect(rect, Paint()..color = color);

void stamp(Canvas canvas, Offset center, double radius, String label) {
  final paint = Paint()
    ..color = stampInk
    ..style = PaintingStyle.stroke
    ..strokeWidth = 3;
  canvas.drawCircle(center, radius, paint);
  canvas.drawCircle(center, radius - 10, paint..strokeWidth = 1.5);
  write(
    canvas,
    label,
    center.dx - radius,
    center.dy - 11,
    size: radius / 3.6,
    weight: FontWeight.w700,
    color: stampInk,
    width: radius * 2,
    align: TextAlign.center,
  );
}

void barcode(Canvas canvas, Rect rect) {
  const widths = [2, 1, 3, 1, 1, 2, 1, 3, 2, 1, 1, 2, 3, 1, 2, 1, 1, 3, 2, 2];
  final paint = Paint()..color = ink;
  var x = rect.left;
  for (var i = 0; x < rect.right; i++) {
    final width = math.min(widths[i % widths.length] * 2.5, rect.right - x);
    if (i.isEven) {
      canvas.drawRect(Rect.fromLTWH(x, rect.top, width, rect.height), paint);
    }
    x += width;
  }
}

/// A table of the paper passport: a header row and ruled rows.
void passportTable(
  Canvas canvas, {
  required double top,
  required List<double> columns,
  required List<String> headers,
  required int rows,
  required double rowHeight,
}) {
  final left = columns.first;
  final right = columns.last;
  final bottom = top + 36 + rows * rowHeight;
  for (var i = 0; i < headers.length; i++) {
    write(
      canvas,
      headers[i],
      columns[i] + 8,
      top + 9,
      size: 13,
      color: inkMuted,
      width: columns[i + 1] - columns[i] - 12,
    );
  }
  for (final x in columns) {
    line(canvas, Offset(x, top), Offset(x, bottom));
  }
  line(canvas, Offset(left, top), Offset(right, top));
  for (var row = 0; row <= rows; row++) {
    final y = top + 36 + row * rowHeight;
    line(canvas, Offset(left, y), Offset(right, y));
  }
}

  pictures['rabies-sticker'] = await picture(800, 600, (canvas) {
    fill(canvas, const Rect.fromLTWH(0, 0, 800, 600), const Color(0xFFE6DCC6));
    for (var y = 40.0; y < 600; y += 90) {

      ..translate(400, 300)
      ..rotate(-0.05);

    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(label, const Radius.circular(16)),
      Paint()..color = const Color(0xFFFFFFFF),
    );
    canvas.drawRRect(
      RRect.fromRectAndCorners(
        Rect.fromLTWH(label.left, label.top, label.width, 72),
        topLeft: const Radius.circular(16),
        topRight: const Radius.circular(16),
      ),
      Paint()..color = const Color(0xFF00857C),
    );
    write(
      canvas,
      'NOBIVAC RABIES',
      label.left + 26,
      label.top + 18,
      size: 32,
      weight: FontWeight.w700,
      color: const Color(0xFFFFFFFF),
      width: 430,
    );
    write(canvas, 'MSD Animal Health', label.left + 26, label.top + 90, size: 20, color: inkMuted);
    write(canvas, 'Lot A123B01', label.left + 26, label.top + 124, size: 27, weight: FontWeight.w700);
    write(
      canvas,
      'Exp ${dmy(d(400)).substring(3)}',
      label.left + 26,
      label.top + 162,
      size: 22,
    );
    barcode(canvas, Rect.fromLTWH(label.left + 26, label.top + 208, 320, 62));
    canvas.restore();
  });

    // look rendered here) and 'Helvetica Neue' (iPhone; Arimo has the same
    // widths, so the layout matches).
    for (final name in ['Arimo', 'Helvetica Neue']) {
      await family(name, [
        for (final weight in ['Regular', 'Medium', 'Bold'])
          asset('Arimo-$weight.ttf'),
      ]);
    }
    await family('JetBrains Mono', [asset('JetBrainsMono-Medium.ttf')]);
    await family('MaterialIcons', [system('MaterialIcons-Regular.otf')]);

        ByteData.sublistView(File(symbolsFont()).readAsBytesSync()),
